<script>
const table = document.getElementById("vocabTable");
let stopFlag = false;
let currentIndex = -1;
let selectedVoice = null;
let darkMode = localStorage.getItem("darkMode") === "true";

function loadVoices() {
const voices = speechSynthesis.getVoices();
selectedVoice =
voices.find(v => v.lang === 'de-DE' && v.name.includes('Natural')) ||
voices.find(v => v.lang === 'de-DE') ||
voices[0];
}
speechSynthesis.onvoiceschanged = loadVoices;

window.onload = function() {
loadVoices();
if (darkMode) document.body.classList.add("dark-mode");

const savedData = JSON.parse(localStorage.getItem("vocabData")) || [];
if(savedData.length) populateTable(savedData);

const lastIndex = parseInt(localStorage.getItem("lastIndex"));
if(!isNaN(lastIndex) && table.rows[lastIndex + 1]) {
setTimeout(() => {
const row = table.rows[lastIndex + 1];
row.classList.add("highlight");
row.scrollIntoView({ behavior: "smooth", block: "center" });
}, 500);
}
}

async function translateWord(word) {
try {
const res = await fetch(`https://api.mymemory.translated.net/get?q=${encodeURIComponent(word)}&langpair=de|ar`);
const data = await res.json();
return data.responseData.translatedText;
} catch {
return "";
}
}

async function addWord() {
const de = engInput.value.trim();
let ar = arInput.value.trim();
if(!de) return alert("اكتب الكلمة بالألماني");
if(!ar) ar = await translateWord(de);

const row = table.insertRow();
row.insertCell(0).innerText = de;
row.insertCell(1).innerText = ar;
row.insertCell(2).innerHTML = `
<button class="stopBtn" onclick="stopSpeaking()">⏸</button>
<button class="restartBtn" onclick="restartSpeaking(${table.rows.length-2})">▶</button>
`;

let saved = JSON.parse(localStorage.getItem("vocabData")) || [];
saved.push({ eng: de, ar });
localStorage.setItem("vocabData", JSON.stringify(saved));

engInput.value = '';
arInput.value = '';
row.scrollIntoView({ behavior: "smooth", block: "center" });
speakWord(de, 1);
}

function populateTable(data) {
while(table.rows.length > 1) table.deleteRow(1);
data.forEach((item, i) => {
const row = table.insertRow();
row.insertCell(0).innerText = item.eng;
row.insertCell(1).innerText = item.ar;
row.insertCell(2).innerHTML = `
<button class="stopBtn" onclick="stopSpeaking()">⏸</button>
<button class="restartBtn" onclick="restartSpeaking(${i})">▶</button>
`;
});
}

function speakSequence(index) {
const data = JSON.parse(localStorage.getItem("vocabData")) || [];
if(index >= data.length || stopFlag) return;

if(currentIndex >= 0)
table.rows[currentIndex +1].classList.remove('highlight');

currentIndex = index;
localStorage.setItem("lastIndex", currentIndex);

const row = table.rows[currentIndex +1];
row.classList.add('highlight');
row.scrollIntoView({ behavior: "smooth", block: "center" });

const word = data[index].eng;
const repeat = parseInt(repeatSelect.value);
stopFlag = false;

speakWord(word, repeat, () => {
if(!stopFlag) speakSequence(index +1);
});
}

function speakWord(word, repeat, onComplete) {
let count = 0;
function speakAgain() {
if(stopFlag) return;
const utter = new SpeechSynthesisUtterance(word);
utter.lang = 'de-DE';
utter.voice = selectedVoice;
utter.onend = () => {
count++;
if(count < repeat) speakAgain();
else if(onComplete) onComplete();
};
speechSynthesis.speak(utter);
}
speakAgain();
}

function stopSpeaking() {
stopFlag = true;
speechSynthesis.cancel();
}

function restartSpeaking(index) {
stopFlag = false;
speechSynthesis.cancel();
setTimeout(() => speakSequence(index), 100);
}

function speakAll() {
stopFlag = false;
speechSynthesis.cancel();
speakSequence(0);
}

function clearStorage() {
localStorage.removeItem("vocabData");
localStorage.removeItem("lastIndex");
while(table.rows.length > 1) table.deleteRow(1);
alert("تم مسح الكلمات.");
}

function toggleDarkMode() {
document.body.classList.toggle("dark-mode");
localStorage.setItem("darkMode", document.body.classList.contains("dark-mode"));
}

function scrollToTop() { window.scrollTo({ top: 0, behavior: 'smooth' }); }
function scrollToBottom() { window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' }); }

function downloadVocab() {
const data = JSON.parse(localStorage.getItem("vocabData")) || [];
if(!data.length) return alert("لا توجد كلمات.");
const content = data.map(i => i.eng + "\n" + i.ar).join("\n");
const a = document.createElement("a");
a.href = URL.createObjectURL(new Blob([content]));
a.download = "vocab.txt";
a.click();
}
document.getElementById("engInput").addEventListener("keydown", function(e) {
    if (e.key === "Enter") {
        e.preventDefault();
        addWord();
    }
});

document.getElementById("arInput").addEventListener("keydown", function(e) {
    if (e.key === "Enter") {
        e.preventDefault();
        addWord();
    }
});
</script>
